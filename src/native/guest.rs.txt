    fn rpc_segments(op:u8, prefix:&[u8], data:&[u8])->Vec<u8> {
        let length=prefix.len().checked_add(data.len()).and_then(|n|n.checked_add(1)).expect("invalid native request size");

        let mut out=std::io::stdout().lock();

    pub(super) fn __gamepkg_next_tick()->Option<(u32,u64)>{
      loop {
        let mut input=std::io::stdin().lock();let mut len=[0;4];if input.read_exact(&mut len).is_err(){return None;}
        let encoded=u32::from_le_bytes(len);let control=encoded&0x80000000!=0;let n=(encoded&0x7fffffff) as usize;
        if control {
            #[cfg(gamepkg_zelda)] {
                assert!((1..=65541).contains(&n));let mut b=vec![0;n];input.read_exact(&mut b).unwrap();drop(input);
                __gamepkg_packet(&engine::snapshot_control(&b));continue;
            }
            #[cfg(not(gamepkg_zelda))] panic!("unsupported native control frame");
        }
        assert!(n>=12&&n<=204&&(n-12)%12==0);
        let mut b=vec![0;n];input.read_exact(&mut b).unwrap();drop(input);
        #[cfg(gamepkg_doom)] engine::input(&b[12..]);
        #[cfg(not(gamepkg_doom))] assert_eq!(n,12);
        return Some((u32::from_le_bytes(b[..4].try_into().unwrap()),u64::from_le_bytes(b[4..12].try_into().unwrap())));
      }
