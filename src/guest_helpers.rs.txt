/// A signed Slang fragment entry; the host compiles it during Build/Rebuild.
pub struct Shader { index: u32 }
impl Shader {
    pub fn from_slang(source: &str, entry: &str) -> Option<Self> {
        shader_index(source, entry).map(|index| Self { index })
    }
    pub fn draw(&self) -> bool { draw_shader(self.index) }
    pub fn draw_rgba(&self, width: u32, height: u32, pixels: &[u8]) -> bool {
        draw_shader_image(self.index, width, height, pixels)
    }
    pub fn draw_texture(&self, path: &str) -> bool { draw_shader_asset_image(self.index, path) }
}
pub const AUDIO_SAMPLE_RATE: u32 = 48000;
pub const MAX_AUDIO_SAMPLES: usize = 1600;
pub fn audio_submit_pcm16_le(bytes: &[u8]) -> bool {
    if bytes.len() % 2 != 0 || bytes.len() > MAX_AUDIO_SAMPLES * 2 { return false; }
    let mut samples = [0i16; MAX_AUDIO_SAMPLES];
    let count = bytes.len() / 2;
    let mut index = 0;
    while index < count {
        samples[index] = i16::from_le_bytes([bytes[index*2], bytes[index*2+1]]);
        index += 1;
    }
    audio_submit(&samples[..count])
}

/// Focused game-surface events. Kinds: reset=0, move=1, down=2, up=3,
/// wheel=4, physical key-down=5/up=6, committed Unicode scalar=7, pointer-reset=8,
/// logical key-down=9/up=10 (SDL keycode; down bit 31 marks repeat).
/// Pointer coordinates use the 640x480 canvas; wheel code is lines=0/pixels=1/pages=2.
/// Key code is an SDL physical scancode (1..511), plus bit 16 on repeated down.
pub struct InputEvent {pub kind:u32,pub x:f32,pub y:f32,pub code:u32}
pub fn input_events()->Option<Vec<InputEvent>> {
    let mut bytes=[0u8;32768];let count=__gamepkg_input_events(&mut bytes)? as usize;
    if count>bytes.len()||count%16!=0{return None;}
    let mut events=Vec::new();
    for record in bytes[..count].chunks_exact(16){events.push(InputEvent{
        kind:u32::from_le_bytes([record[0],record[1],record[2],record[3]]),
        x:f32::from_le_bytes([record[4],record[5],record[6],record[7]]),
        y:f32::from_le_bytes([record[8],record[9],record[10],record[11]]),
        code:u32::from_le_bytes([record[12],record[13],record[14],record[15]])});}
    Some(events)
}

/// One player-selected file operation. Status: pending=1, imported=2,
/// exported=3, cancelled=4, failed=5. No host paths or directory access.
pub struct UserFileStatus {pub status:u32,pub size:u32,pub name_bytes:u32}
pub fn file_import(index:u32)->Option<u32> {
    let mut out=[0u8;4];let n=__gamepkg_file_call(0,&u32::to_le_bytes(index),&mut out)?;
    if n!=4 {None}else{Some(u32::from_le_bytes(out))}
}
pub fn file_export(index:u32,name:&str,bytes:&[u8])->Option<u32> {
    if name.len()>256||bytes.len()>1048576{return None;}
    let mut request=Vec::new();request.extend_from_slice(&u32::to_le_bytes(index));request.extend_from_slice(&u32::to_le_bytes(name.len() as u32));request.extend_from_slice(name.as_bytes());request.extend_from_slice(bytes);
    let mut out=[0u8;4];let n=__gamepkg_file_call(1,&request,&mut out)?;if n!=4 {None}else{Some(u32::from_le_bytes(out))}
}
pub fn file_status(handle:u32)->Option<UserFileStatus> {
    let mut out=[0u8;12];if __gamepkg_file_call(2,&u32::to_le_bytes(handle),&mut out)?!=12{return None;}
    Some(UserFileStatus{status:u32::from_le_bytes([out[0],out[1],out[2],out[3]]),size:u32::from_le_bytes([out[4],out[5],out[6],out[7]]),name_bytes:u32::from_le_bytes([out[8],out[9],out[10],out[11]])})
}
pub fn file_name(handle:u32,output:&mut[u8])->Option<u32>{__gamepkg_file_call(3,&u32::to_le_bytes(handle),output)}
pub fn file_read(handle:u32,offset:u32,output:&mut[u8])->Option<u32>{let mut request=Vec::new();request.extend_from_slice(&u32::to_le_bytes(handle));request.extend_from_slice(&u32::to_le_bytes(offset));__gamepkg_file_call(4,&request,output)}
pub fn file_close(handle:u32)->bool{__gamepkg_file_call(5,&u32::to_le_bytes(handle),&mut[])==Some(0)}
