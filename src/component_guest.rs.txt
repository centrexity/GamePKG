    unsafe fn copy_result(result: [u32;3], output: &mut [u8]) -> Option<u32> {
        if result[0] == 0 { return None; }
        let n = result[2] as usize;
        let pointer = result[1] as *mut u8;
        if n > output.len() { unreachable!(); }
        if n > 0 {
            output[..n].copy_from_slice(unsafe { std::slice::from_raw_parts(pointer,n) });
            unsafe { std::alloc::dealloc(pointer,std::alloc::Layout::from_size_align(n,1).unwrap()) };
        }
        Some(result[2])
    }

#[unsafe(no_mangle)]
pub unsafe extern "C" fn cabi_realloc(old: *mut u8, old_size: usize, align: usize, new_size: usize) -> *mut u8 {
    if new_size == 0 { return align as *mut u8; }
    let layout = std::alloc::Layout::from_size_align(new_size,align).unwrap();
    let pointer = if old_size == 0 { unsafe { std::alloc::alloc(layout) } } else {
        unsafe { std::alloc::realloc(old,std::alloc::Layout::from_size_align(old_size,align).unwrap(),new_size) }
    };
    if pointer.is_null() { std::alloc::handle_alloc_error(layout); }
    pointer
}
